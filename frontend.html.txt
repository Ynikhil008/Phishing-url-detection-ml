<script>
async function checkURL() {
    const url = document.getElementById("urlInput").value;

    if (!url) {
        alert("Please enter a URL");
        return;
    }

    try {
        const response = await fetch("http://127.0.0.1:8000/predict/url", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({ url: url })
        });

        const data = await response.json();

        let resultHTML = `
            Prediction: <span class="${data.prediction === 'PHISHING' ? 'phishing' : 'legit'}">${data.prediction}</span><br>
            Confidence: ${data.confidence}<br>
            Risk Level: ${data.risk_level}<br>
        `;

        if (data.suspicious_signals && data.suspicious_signals.length > 0) {
            resultHTML += "<br>⚠️ Signals:<ul>";
            data.suspicious_signals.forEach(signal => {
                resultHTML += `<li>${signal}</li>`;
            });
            resultHTML += "</ul>";
        }

        document.getElementById("result").innerHTML = resultHTML;

    } catch (error) {
        document.getElementById("result").innerHTML = "❌ Error connecting to backend";
    }
}
</script>
